# controller/agentic_loop.py
# ─────────────────────────────────────────────────────────────────
# The self-correcting agentic loop.
# Replaces ToolOrchestrator. Import this in query_controller.py.
# ─────────────────────────────────────────────────────────────────

from dataclasses import dataclass, field
from typing import Any
import logging

from controller.loop_state import LoopState
from controller.self_corrector import SelfCorrector
from model.facade import ModelFacade
from tools import ALL_TOOLS

log = logging.getLogger(__name__)

class AgenticLoop:
    """
    Self-correcting agentic loop. Runs up to max_iterations.
    On tool call  → dispatches to metadata layer, appends to state.
    On emit SQL   → validates, self-corrects if invalid, executes.
    On runtime err → re-prompts LLM with error context, retries.
    """

    def __init__(
        self,
        model: ModelFacade,
        max_iterations: int = 5,
        max_correction_attempts: int = 2,
    ):
        self.model = model
        self.max_iter = max_iterations
        self.max_corrections = max_correction_attempts
        self.corrector = SelfCorrector()

    async def run(self, query: str) -> LoopState:
        state = LoopState(nl_query=query)
        messages = [self._build_system_msg(query)]

        for i in range(self.max_iter):
            state.iteration = i + 1
            log.info(f"Loop iteration {state.iteration}")

            # ── LLM reasons over current state ──
            response = await self._call_llm(messages, state)

            if response.stop_reason == "tool_use":
                # ── Tool call branch ──
                results = await self._dispatch_tools(response.tool_calls)
                state.tool_results.extend(results)
                messages = self._append_tool_results(messages, response, results)

            elif response.stop_reason == "end_turn":
                # ── SQL emit branch ──
                sql = self._extract_sql(response)
                if not sql:
                    break

                # ── SQL validation gate ──
                val_result = self.model.validate_sql(sql)
                if not val_result.valid:
                    corrected = await self.corrector.correct(
                        sql, val_result.error, state, self.max_corrections
                    )
                    if corrected:
                        sql = corrected
                    else:
                        state.add_error("validation", val_result.error)
                        messages = self._append_error(messages, val_result.error)
                        continue   # retry iteration

                # ── Execute against DB ──
                exec_result = await self.model.db.execute(sql)
                if exec_result.error:
                    # runtime error → re-prompt with error context
                    state.add_error("runtime", exec_result.error)
                    messages = self._append_runtime_error(
                        messages, sql, exec_result.error
                    )
                    continue   # retry iteration with error in context

                # ── Success ──
                state.final_sql = sql
                state.rows = exec_result.rows
                state.confidence = self._score_confidence(state)
                state.success = True
                break

        state.finalize()
        return state

from controller.agentic_loop import AgenticLoop
from controller.result_formatter import ResultFormatter
from model.facade import ModelFacade
from view.schemas.request import QueryRequest
from view.schemas.response import QueryResponse

class QueryController:
    def __init__(self, model: ModelFacade, settings):
        self.loop = AgenticLoop(
            model=model,
            max_iterations=settings.loop_max_iter,       # default 5
            max_correction_attempts=settings.max_corrections, # default 2
        )
        self.formatter = ResultFormatter()

    async def handle(self, req: QueryRequest) -> QueryResponse:
        # Run the full agentic loop
        state = await self.loop.run(req.nl_query)

        if not state.success:
            raise QueryFailedError(
                f"Loop exhausted after {state.iteration} iterations. "
                f"Errors: {state.errors}"
            )

        return self.formatter.format(state)

# controller/loop_state.py
# Accumulates all state across iterations.
# Passed to LLM as context on every round.

from dataclasses import dataclass, field
from datetime import datetime, timezone
from typing import Any

@dataclass
class LoopState:
    nl_query: str
    iteration: int = 0
    tool_results: list[dict] = field(default_factory=list)
    errors: list[dict] = field(default_factory=list)
    sql_attempts: list[str] = field(default_factory=list)
    final_sql: str | None = None
    rows: list[dict] = field(default_factory=list)
    confidence: float = 0.0
    success: bool = False
    started_at: datetime = field(default_factory=lambda: datetime.now(timezone.utc))
    finished_at: datetime | None = None
    token_count: int = 0

    def add_error(self, kind: str, msg: str):
        self.errors.append({
            "kind": kind,
            "message": msg,
            "iteration": self.iteration,
        })

    def finalize(self):
        self.finished_at = datetime.now(timezone.utc)
        self.token_count = sum(
            len(str(r)) for r in self.tool_results
        )

    def to_context_dict(self) -> dict:
        """Serialised for LLM prompt injection."""
        return {
            "nl_query": self.nl_query,
            "iteration": self.iteration,
            "tool_results": self.tool_results[-10:],   # last 10
            "errors": self.errors,
            "sql_attempts": self.sql_attempts,
        }

# controller/self_corrector.py
# Called when SQL validation fails.
# Re-prompts the LLM with the invalid SQL + error message,
# asking it to return a corrected version.

class SelfCorrector:
    async def correct(
        self,
        bad_sql: str,
        error: str,
        state: LoopState,
        max_attempts: int = 2,
    ) -> str | None:
        for attempt in range(max_attempts):
            prompt = self._correction_prompt(bad_sql, error, state)
            response = await llm.call([{
                "role": "user",
                "content": prompt,
            }])
            corrected = self._extract_sql(response)
            if corrected:
                state.sql_attempts.append(corrected)
                return corrected

        return None   # correction exhausted

    def _correction_prompt(self, sql, error, state) -> str:
        return f"""
The following SQL is invalid. Fix it and return only corrected SQL.

Original NL query: {state.nl_query}

Invalid SQL:
{sql}

Validation error:
{error}

Schema context collected so far:
{state.tool_results}

Return ONLY the corrected SQL, no explanation.
"""

04 — Contracts
Loop rules
These are non-negotiable constraints for the agentic loop. Breaking them causes silent wrong answers, infinite loops, or unsafe DB execution.
✗
Never execute SQL without validation passing first. SQLValidator must run inside the loop before every db.execute(). Self-correction happens here, not after execution.
✗
Never run the loop without a hard iteration ceiling. Always enforce max_iterations. An uncapped loop on a hard query will exhaust tokens and never return. Default is 5; never set it above 10.
✗
Never silently swallow a failed loop. If the loop exits without state.success = True, raise a QueryFailedError with the full error list. The View layer surfaces this as a structured 422.
!
Limit self-correction to 2 attempts per SQL emit. More than 2 correction attempts on the same SQL means the schema context is wrong — the loop should re-collect context (tool calls) rather than keep correcting bad SQL.
!
Append all errors to state before re-entering the loop. The LLM must see previous errors in its context window each iteration. LoopState.errors is serialised into every subsequent message. If errors are not injected, the LLM repeats the same mistake.
!
Always include a row limit guard in the executor. Wrap every SELECT with LIMIT 10000 unless the query explicitly requests aggregate output. Runtime timeout on large tables causes the loop to retry unnecessarily.
✓
Log every iteration to QueryLog. Write (nl_query, iteration, sql_attempt, error, success) for every loop cycle. This data trains future few-shot retrieval and reveals which query types cause the most correction cycles.
✓
Return confidence score with every response. Compute it from: iterations used, correction count, row plausibility, schema coverage ratio. Low-confidence responses should be flagged in the API response so the caller can decide whether to show them.
